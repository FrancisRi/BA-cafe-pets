add_custom_command(TARGET ba_cafe_lite_native POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:ba_cafe_lite_native>/model" "$<TARGET_FILE_DIR:ba_cafe_lite_native>/behavior" "$<TARGET_FILE_DIR:ba_cafe_lite_native>/licenses"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${BA_ROOT}/assets/Shiroko.glb" "$<TARGET_FILE_DIR:ba_cafe_lite_native>/model/Shiroko.glb"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/release/defaults.ini" "$<TARGET_FILE_DIR:ba_cafe_lite_native>/defaults.ini"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${BA_ROOT}/design/template.bt.xml" "$<TARGET_FILE_DIR:ba_cafe_lite_native>/behavior/template.bt.xml"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${BA_ROOT}/design/battle-mode.drawio" "${BA_ROOT}/design/battle-mode.mmd" "$<TARGET_FILE_DIR:ba_cafe_lite_native>/behavior"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${raylib_SOURCE_DIR}/LICENSE" "$<TARGET_FILE_DIR:ba_cafe_lite_native>/licenses/raylib.txt"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${raylib_SOURCE_DIR}/src/external/glfw/LICENSE.md" "$<TARGET_FILE_DIR:ba_cafe_lite_native>/licenses/GLFW.txt"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/licenses" "$<TARGET_FILE_DIR:ba_cafe_lite_native>/licenses"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${behaviortree_cpp_SOURCE_DIR}/LICENSE" "$<TARGET_FILE_DIR:ba_cafe_lite_native>/licenses/BehaviorTree.CPP.txt"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/README.md" "$<TARGET_FILE_DIR:ba_cafe_lite_native>/README.md"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/CREDITS.md" "${CMAKE_CURRENT_SOURCE_DIR}/NOTICE.md" "$<TARGET_FILE_DIR:ba_cafe_lite_native>"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE.md" "$<TARGET_FILE_DIR:ba_cafe_lite_native>"
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:ba_cafe_lite_native>/docs"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/docs/LICENSE-SCOPE.md" "${CMAKE_CURRENT_SOURCE_DIR}/docs/DEVELOPMENT.md" "$<TARGET_FILE_DIR:ba_cafe_lite_native>/docs"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/release/README.txt" "${CMAKE_CURRENT_SOURCE_DIR}/release/CHANGELOG.md" "$<TARGET_FILE_DIR:ba_cafe_lite_native>")

endforeach()
